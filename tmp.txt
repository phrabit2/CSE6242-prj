#!/usr/bin/env python3
"""Label Studio JSON -> Y/N 분류용 학습 데이터(train.jsonl) 변환.
자원 벤치마크용. 20장 전부 학습에 사용."""
import json, os, re, sys

ANN   = "/data/sample_train/data/annotations.json"
IMGDIR = "/data/sample_train/data/images"
OUT   = "/data/sample_train/data/train.jsonl"

QUESTION = "이 상자 이미지에 파손이 있습니까? '예' 또는 '아니오'로만 답하세요."

def candidates(task):
    """이미지 파일명 후보들을 생성 (LS 해시 접두어 대응)."""
    names = []
    fu = task.get("file_upload")
    if fu:
        names.append(fu)                                  # 해시 붙은 원래 이름
        names.append(re.sub(r'^[0-9a-fA-F]{8}-', '', fu)) # 해시 접두어 제거
    img = task.get("data", {}).get("image", "")
    if img:
        base = os.path.basename(img)
        names.append(base)
        names.append(re.sub(r'^[0-9a-fA-F]{8}-', '', base))
    # 중복 제거(순서 유지)
    seen, out = set(), []
    for n in names:
        if n and n not in seen:
            seen.add(n); out.append(n)
    return out

def resolve(task, existing):
    for c in candidates(task):
        if c in existing:
            return c
    return None

def has_damage(task):
    for ann in task.get("annotations", []):
        for r in ann.get("result", []):
            labels = r.get("value", {}).get("rectanglelabels", [])
            if labels:
                return True
    return False

def main():
    data = json.load(open(ANN))
    existing = set(os.listdir(IMGDIR))
    samples, unmatched = [], []
    for task in data:
        fname = resolve(task, existing)
        if fname is None:
            unmatched.append(candidates(task)[:1])
            continue
        answer = "예" if has_damage(task) else "아니오"
        samples.append({
            "image": os.path.join(IMGDIR, fname),
            "conversations": [
                {"role": "user",      "content": "<image>\n" + QUESTION},
                {"role": "assistant", "content": answer},
            ],
        })

    with open(OUT, "w", encoding="utf-8") as f:
        for s in samples:
            f.write(json.dumps(s, ensure_ascii=False) + "\n")

    n_yes = sum(1 for s in samples if s["conversations"][1]["content"] == "예")
    print(f"총 task: {len(data)}")
    print(f"매칭 성공: {len(samples)}  (예:{n_yes}  아니오:{len(samples)-n_yes})")
    print(f"매칭 실패: {len(unmatched)}")
    if unmatched:
        print("  실패 예시:", unmatched[:3])
        print("  -> images 폴더 파일명과 JSON의 file_upload를 대조 필요")
    print(f"출력: {OUT}")

if __name__ == "__main__":
    main()
