
sudo docker run --rm \
  --network pinpoint-docker_pinpoint \
  --entrypoint bash \
  -v /data/pinpoint/pinot-schemas:/schemas:ro \
  apachepinot/pinot:1.0.0-11-amazoncorretto \
  -c 'cd /tmp && cp /schemas/*.json . && sed -i "s/localhost:19092/pinpoint-kafka:9092/g" *.json &&
  for t in uriStat tag double dataType exceptionTrace inspectorAgent inspectorApplication; do
    echo "=== AddTable: $t ===";
    /opt/pinot/bin/pinot-admin.sh AddTable -schemaFile ${t}Schema.json -tableConfigFile ${t}Table.json -controllerHost pinot-controller -controllerPort 9000 -exec || true;
  done'
