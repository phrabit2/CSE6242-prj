cat > /tmp/addtables.sh <<'EOF'
#!/bin/bash
cd /tmp/schemas
sed -i 's/localhost:19092/pinpoint-kafka:9092/g' *.json
sed -i 's/"replicasPerPartition": *"3"/"replicasPerPartition": "2"/g' *.json
sed -i 's/"replication": *"3"/"replication": "2"/g' *.json
for t in uriStat tag double dataType inspectorAgent inspectorApplication; do
  echo "=== AddTable: $t ===";
  /opt/pinot/bin/pinot-admin.sh AddTable -schemaFile ${t}Schema.json -tableConfigFile ${t}Table.json -controllerHost pinot-controller -controllerPort 9000 -exec 2>&1 | grep -E "successfully|error|Exception";
done
EOF

# 스키마 + 스크립트를 컨테이너에 복사
sudo docker cp /data/pinpoint/pinot-schemas pinpoint-docker-pinot-server-0-1:/tmp/schemas
sudo docker cp /tmp/addtables.sh pinpoint-docker-pinot-server-0-1:/tmp/addtables.sh

# 컨테이너 안에서 실행
sudo docker exec pinpoint-docker-pinot-server-0-1 bash /tmp/addtables.sh
