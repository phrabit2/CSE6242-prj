def has_damage(task):
    for ann in task.get("annotations", []):
        for r in ann.get("result", []):
            labels = r.get("value", {}).get("rectanglelabels", [])
            if "damaged area" in labels:   # product area는 무시, damaged area만 카운트
                return True
    return False

#!/usr/bin/env python3
"""Qwen2.5-VL LoRA 파인튜닝 (자원 벤치마크용) - batch size 실험 버전."""
import os, sys, json, argparse
import torch
from torch.utils.data import Dataset, DataLoader
from PIL import Image
from transformers import Qwen2_5_VLForConditionalGeneration, AutoProcessor
from peft import LoraConfig, get_peft_model

sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
from bench import Bench

# ---------- 인자 ----------
parser = argparse.ArgumentParser()
parser.add_argument("--batch_size", type=int, default=1)
parser.add_argument("--epochs", type=int, default=3)
args = parser.parse_args()

MODEL_PATH = "/data/ai-models/Qwen2.5-VL-7B-Instruct"
DATA       = "/data/sample_train/data/train.jsonl"
METHOD     = f"lora_b{args.batch_size}"          # 예: lora_b3
OUTDIR     = f"/data/sample_train/outputs/{METHOD}"

BATCH_SIZE = args.batch_size
EPOCHS     = args.epochs
LR         = 1e-4
MAX_PIXELS = 512*28*28
GRAD_CKPT  = True
SEED       = 42
LORA_R, LORA_ALPHA, LORA_DROPOUT = 16, 32, 0.05
# --------------------------

torch.manual_seed(SEED)

def load_data():
    with open(DATA) as f:
        return [json.loads(l) for l in f]

class VLDataset(Dataset):
    def __init__(self, rows):
        self.rows = rows
    def __len__(self):
        return len(self.rows)
    def __getitem__(self, i):
        r = self.rows[i]
        img = Image.open(r["image"]).convert("RGB")
        user_text = r["conversations"][0]["content"].replace("<image>\n", "")
        answer    = r["conversations"][1]["content"]
        return {
            "messages": [
                {"role": "user", "content": [
                    {"type": "image", "image": img},
                    {"type": "text",  "text": user_text}]},
                {"role": "assistant", "content": [{"type": "text", "text": answer}]},
            ],
            "answer": answer,
        }

def collate(batch, processor):
    """batch>1 지원. 각 샘플을 개별 토큰화 후 패딩으로 묶음."""
    input_ids_list, labels_list, pixel_list, grid_list = [], [], [], []

    for sample in batch:
        messages = sample["messages"]
        full = processor.apply_chat_template(
            messages, tokenize=True, add_generation_prompt=False,
            return_dict=True, return_tensors="pt")
        prompt_only = processor.apply_chat_template(
            messages[:-1], tokenize=True, add_generation_prompt=True,
            return_dict=True, return_tensors="pt")

        ids = full["input_ids"][0]
        labels = ids.clone()
        plen = prompt_only["input_ids"].shape[1]
        labels[:plen] = -100

        input_ids_list.append(ids)
        labels_list.append(labels)
        pixel_list.append(full["pixel_values"])
        grid_list.append(full["image_grid_thw"])

    # 텍스트 패딩 (오른쪽)
    pad_id = processor.tokenizer.pad_token_id or 0
    maxlen = max(x.size(0) for x in input_ids_list)
    input_ids, labels, attn = [], [], []
    for ids, lbl in zip(input_ids_list, labels_list):
        padlen = maxlen - ids.size(0)
        input_ids.append(torch.cat([ids, torch.full((padlen,), pad_id, dtype=ids.dtype)]))
        labels.append(torch.cat([lbl, torch.full((padlen,), -100, dtype=lbl.dtype)]))
        attn.append(torch.cat([torch.ones(ids.size(0), dtype=torch.long),
                               torch.zeros(padlen, dtype=torch.long)]))

    return {
        "input_ids": torch.stack(input_ids),
        "attention_mask": torch.stack(attn),
        "labels": torch.stack(labels),
        "pixel_values": torch.cat(pixel_list, dim=0),
        "image_grid_thw": torch.cat(grid_list, dim=0),
    }

def main():
    os.makedirs(OUTDIR, exist_ok=True)
    bench = Bench(METHOD)

    processor = AutoProcessor.from_pretrained(
        MODEL_PATH, max_pixels=MAX_PIXELS, trust_remote_code=True)
    model = Qwen2_5_VLForConditionalGeneration.from_pretrained(
        MODEL_PATH, torch_dtype=torch.bfloat16,
        device_map="cuda:0", trust_remote_code=True)

    lora_cfg = LoraConfig(
        r=LORA_R, lora_alpha=LORA_ALPHA, lora_dropout=LORA_DROPOUT,
        target_modules=["q_proj","k_proj","v_proj","o_proj",
                        "gate_proj","up_proj","down_proj"],
        task_type="CAUSAL_LM")
    model = get_peft_model(model, lora_cfg)

    if GRAD_CKPT:
        model.gradient_checkpointing_enable()
        model.enable_input_require_grads()
    model.config.use_cache = False

    bench.mark_model(model)
    model.print_trainable_parameters()

    rows = load_data()
    ds = VLDataset(rows)
    loader = DataLoader(ds, batch_size=BATCH_SIZE, shuffle=True,
                        collate_fn=lambda b: collate(b, processor))

    optim = torch.optim.AdamW(
        [p for p in model.parameters() if p.requires_grad], lr=LR)

    model.train()
    device = next(model.parameters()).device
    step = 0
    bench.start_train()
    for epoch in range(EPOCHS):
        for batch in loader:
            batch = {k: v.to(device) for k, v in batch.items()}
            loss = model(**batch).loss
            loss.backward()
            optim.step(); optim.zero_grad()
            step += 1
            print(f"[{METHOD}] epoch {epoch+1} step {step} loss {loss.item():.4f}")
    bench.end_train(num_samples=len(rows)*EPOCHS, num_steps=step)

    model.save_pretrained(OUTDIR)
    processor.save_pretrained(OUTDIR)
    print(f"어댑터 저장: {OUTDIR}")
    bench.dump()

if __name__ == "__main__":
    main()
