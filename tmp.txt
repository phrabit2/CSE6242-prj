#!/usr/bin/env python3
"""자원·시간 측정 유틸. 세 학습 스크립트(full/lora/qlora) 공용.
사용법:
    from bench import Bench
    bench = Bench("qlora")
    bench.mark_model(model)          # 모델 로드 직후 1회
    bench.start_train()              # 학습 루프 시작 직전
    ... 학습 ...
    bench.end_train(num_samples=20, num_steps=steps)
    bench.dump()                     # results.csv 에 한 줄 append
"""
import time, csv, os, torch

RESULTS = "/data/sample_train/results.csv"

class Bench:
    def __init__(self, method):
        self.method = method
        self.rec = {"method": method}
        torch.cuda.reset_peak_memory_stats()
        torch.cuda.empty_cache()
        # 모델 로드 전 baseline VRAM
        self.rec["vram_baseline_gb"] = round(torch.cuda.memory_allocated()/1e9, 3)

    def mark_model(self, model):
        total = sum(p.numel() for p in model.parameters())
        trainable = sum(p.numel() for p in model.parameters() if p.requires_grad)
        self.rec["total_params_M"]     = round(total/1e6, 1)
        self.rec["trainable_params_M"] = round(trainable/1e6, 1)
        self.rec["trainable_pct"]      = round(100*trainable/total, 3)
        self.rec["vram_after_load_gb"] = round(torch.cuda.memory_allocated()/1e9, 3)

    def start_train(self):
        torch.cuda.synchronize()
        self._t0 = time.time()

    def end_train(self, num_samples, num_steps):
        torch.cuda.synchronize()
        dt = time.time() - self._t0
        self.rec["total_time_s"]     = round(dt, 2)
        self.rec["num_steps"]        = num_steps
        self.rec["sec_per_step"]     = round(dt/max(num_steps,1), 3)
        self.rec["samples_per_sec"]  = round(num_samples/dt, 3) if dt>0 else 0
        self.rec["peak_vram_alloc_gb"]    = round(torch.cuda.max_memory_allocated()/1e9, 3)
        self.rec["peak_vram_reserved_gb"] = round(torch.cuda.max_memory_reserved()/1e9, 3)

    def dump(self):
        cols = ["method","total_params_M","trainable_params_M","trainable_pct",
                "vram_baseline_gb","vram_after_load_gb",
                "peak_vram_alloc_gb","peak_vram_reserved_gb",
                "total_time_s","num_steps","sec_per_step","samples_per_sec"]
        exists = os.path.exists(RESULTS)
        with open(RESULTS, "a", newline="") as f:
            w = csv.DictWriter(f, fieldnames=cols)
            if not exists:
                w.writeheader()
            w.writerow({c: self.rec.get(c, "") for c in cols})
        print("\n=== BENCH RESULT ({}) ===".format(self.method))
        for c in cols:
            print(f"  {c:24s}: {self.rec.get(c,'')}")
        print(f"=> appended to {RESULTS}")
















#!/usr/bin/env bash
# 사용법: ./monitor.sh <method>   (예: ./monitor.sh qlora)
# 학습 시작 직전 백그라운드로 실행하고, 학습 끝나면 stop_monitor.sh 로 종료
METHOD="${1:-run}"
OUTDIR="/data/sample_train/outputs/${METHOD}"
mkdir -p "$OUTDIR"

# GPU 시계열
nvidia-smi --query-gpu=timestamp,memory.used,memory.total,utilization.gpu,power.draw \
  --format=csv,nounits -l 1 > "${OUTDIR}/gpu_log.csv" &
echo $! > "${OUTDIR}/.gpu_pid"

# CPU/RAM 시계열 (epoch초, CPU%, 사용RAM MB)
( while true; do
    ts=$(date +%s)
    cpu=$(top -bn1 | awk '/%Cpu/{print $2}')
    memused=$(free -m | awk '/Mem:/{print $3}')
    echo "${ts},${cpu},${memused}"
    sleep 1
  done ) > "${OUTDIR}/cpu_log.csv" &
echo $! > "${OUTDIR}/.cpu_pid"

echo "모니터링 시작 (method=${METHOD}) -> ${OUTDIR}/{gpu,cpu}_log.csv"
echo "종료하려면: /data/sample_train/src/stop_monitor.sh ${METHOD}"














#!/usr/bin/env bash
METHOD="${1:-run}"
OUTDIR="/data/sample_train/outputs/${METHOD}"
for p in gpu cpu; do
  if [ -f "${OUTDIR}/.${p}_pid" ]; then
    kill "$(cat ${OUTDIR}/.${p}_pid)" 2>/dev/null
    rm -f "${OUTDIR}/.${p}_pid"
  fi
done
echo "모니터링 종료 (method=${METHOD})"
