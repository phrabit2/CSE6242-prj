$files = Get-ChildItem -Filter *.jpg | Sort-Object Name
for ($i = 0; $i -lt $files.Count; $i++) {
    if ($files[$i].Name -like '*_annotated.jpg') {
        # annotated 본인 복사
        Copy-Item $files[$i].FullName -Destination damaged
        # 정렬상 바로 앞 파일 복사 (존재할 때만)
        if ($i -gt 0) {
            Copy-Item $files[$i-1].FullName -Destination damaged
        }
    }
}
