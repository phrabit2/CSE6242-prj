#!/usr/bin/env python3
"""train.jsonl (커스텀 형식) → LLaMA-Factory sharegpt 멀티모달 형식 변환."""
import json

SRC = "/data/sample_train/data/train.jsonl"
DST = "/data/sample_train/data/llamafactory_train.json"

out = []
with open(SRC) as f:
    for line in f:
        r = json.loads(line)
        # 이미지 경로 분리
        image_path = r["image"]
        # conversations 그대로 messages로 (content의 <image>\n 유지)
        messages = []
        for turn in r["conversations"]:
            messages.append({
                "role": turn["role"],
                "content": turn["content"],
            })
        out.append({
            "messages": messages,
            "images": [image_path],
        })

# LLaMA-Factory는 JSON 배열 파일을 읽음 (jsonl 아님)
with open(DST, "w", encoding="utf-8") as f:
    json.dump(out, f, ensure_ascii=False, indent=2)

print(f"변환 완료: {len(out)}개")
print(f"출력: {DST}")
print("--- 샘플 ---")
print(json.dumps(out[0], ensure_ascii=False, indent=2))
